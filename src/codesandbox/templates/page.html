{% app_script "landing-runtime-tabs" %}
(function(){
  var root = document.querySelector('[data-runtime-tabs]');
  if(!root) return;
  var tabs = root.querySelectorAll('[data-runtime-tab]');
  var panels = root.querySelectorAll('[data-runtime-panel]');
  tabs.forEach(function(tab){
    tab.addEventListener('click', function(){
      tabs.forEach(function(t){ t.removeAttribute('aria-current'); t.classList.remove('bg-stone-50', 'dark:bg-stone-950'); t.classList.add('bg-white', 'dark:bg-stone-900'); });
      tab.setAttribute('aria-current', 'true');
      tab.classList.remove('bg-white', 'dark:bg-stone-900');
      tab.classList.add('bg-stone-50', 'dark:bg-stone-950');
      var key = tab.getAttribute('data-runtime-tab');
      panels.forEach(function(panel){
        var active = panel.getAttribute('data-runtime-panel') === key;
        panel.classList.toggle('opacity-100', active);
        panel.classList.toggle('translate-y-0', active);
        panel.classList.toggle('pointer-events-auto', active);
        panel.classList.toggle('opacity-0', !active);
        panel.classList.toggle('translate-y-2', !active);
        panel.classList.toggle('pointer-events-none', !active);
      });
    });
  });
})();
{% end_app_script %}
