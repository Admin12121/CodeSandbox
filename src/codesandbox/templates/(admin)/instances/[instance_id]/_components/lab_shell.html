    function saveFile(path, options) {
      options = options || {};
      var state = openFileState[path];
      if (!state || !state.view) return Promise.resolve();

      var content = typeof state.view.getValue === "function"
        ? state.view.getValue()
        : state.view.state.doc.toString();
      if (content === state.original) {
        setDirty(path, false);
        return Promise.resolve();
      }
      var editorStatus = $("[data-editor-status]");
      if (editorStatus) editorStatus.textContent = "Saving…";
      return fsWrite(path, content).then(function (res) {
        if (res.ok) {
          state.original = content;
          setDirty(path, false);
          if (editorStatus) editorStatus.textContent = "Saved";
          window.setTimeout(function () { if (editorStatus) editorStatus.textContent = ""; }, 1200);
          if (!options.quiet) window.Toast && window.Toast.create({ title: "Saved", type: "success", duration: 1500 });
        } else {
          if (editorStatus) editorStatus.textContent = "Save failed";
          window.Toast && window.Toast.create({ title: "Save failed", description: res.error || "", type: "error" });
        }
      }).catch(function (err) {
        if (editorStatus) editorStatus.textContent = "Save failed";
        window.Toast && window.Toast.create({ title: "Save failed", description: err.message, type: "error" });
      });
    }

    function scheduleFileAutosave(path) {
      window.clearTimeout(editorSaveTimers[path]);
      editorSaveTimers[path] = window.setTimeout(function () {
        saveFile(path, { quiet: true });
      }, 700);
    }

    function saveActiveFile() {
      var activeTab = $$("[data-tab]").find(function (el) { return el.classList.contains("active-tab"); });
      if (!activeTab) return;
      window.clearTimeout(editorSaveTimers[activeTab.dataset.tab]);
      saveFile(activeTab.dataset.tab);
    }

    function togglePanel(name) {
      var el = panel(name);
      if (!el) return;

      var visible = hidden(el);

      if (name === "terminal" && visible) {
        show(el);
        syncHandle(el, true);
        setActive('[data-toggle-panel="terminal"]', true);

        var terminalName = ensureTerminalTab();
        if (terminalName) openTerminal(terminalName);

        syncCenterLayout();
        window.dispatchEvent(new Event("resize"));
        return;
      }

      visible ? show(el) : hide(el);
      syncHandle(el, visible);
      setActive('[data-toggle-panel="' + name + '"]', visible);

      if (name === "terminal") {
        syncCenterLayout();
      }

      window.dispatchEvent(new Event("resize"));
    }

    function openRight(mode) {
      var right = panel("right");
      show(right);
      syncHandle(right, true);

      $$("[data-right-pane]").forEach(function (el) {
        el.classList.toggle("hidden", el.dataset.rightPane !== mode);
      });

      $$("[data-toggle-right]").forEach(function (btn) {
        var active = btn.dataset.toggleRight === mode;
        btn.classList.toggle("active-status", active);
        btn.classList.toggle("text-accent-foreground", active);
        btn.classList.toggle("text-muted-foreground", !active);
      });

      window.dispatchEvent(new Event("resize"));
    }

    function toggleRight(mode) {
      var right = panel("right");
      var pane = $('[data-right-pane="' + mode + '"]');

      if (right && !hidden(right) && pane && !hidden(pane)) {
        hide(right);
        syncHandle(right, false);
        setActive("[data-toggle-right]", false);
        window.dispatchEvent(new Event("resize"));
        return;
      }

      openRight(mode);
    }

    function openEditor(file) {
      var pane = editorPaneByPath(file);
      var tab = editorTabByPath(file);

      if (!pane || !tab) return;

      tab.classList.remove("hidden");
      tab.classList.add("flex");

      $$("[data-editor-file]").forEach(function (el) {
        el.classList.toggle("hidden", el.dataset.editorFile !== file);
      });

      $$("[data-tab]").forEach(function (el) {
        var active = el.dataset.tab === file;
        el.classList.toggle("active-tab", active);
        el.classList.toggle("text-muted-foreground", !active);
      });

      var title = $("#current-file-title");
      if (title) title.textContent = file;

      syncCenterLayout();
      window.dispatchEvent(new Event("resize"));
      var state = openFileState[file];
      if (state && state.view && typeof state.view.layout === "function") {
        window.setTimeout(function () { state.view.layout(); }, 0);
      }
    }

    function closeEditor(file) {
      var tab = editorTabByPath(file);
      var pane = editorPaneByPath(file);

      if (!tab) return;

      var wasActive = tab.classList.contains("active-tab");

      var state = openFileState[file];
      if (state && state.view) { try { state.view.destroy(); } catch (e) {} }
      window.clearTimeout(editorSaveTimers[file]);
      delete editorSaveTimers[file];
      delete openFileState[file];

      tab.remove();
      if (pane) pane.remove();

      if (wasActive) {
        var next = visibleEditorTabs()[0];

        if (next) {
          openEditor(next.dataset.tab);
          return;
        }

        var title = $("#current-file-title");
        if (title) title.textContent = "";
      }

      syncCenterLayout();
      window.dispatchEvent(new Event("resize"));
    }

    function openTerminal(name) {
      var terminal = panel("terminal");
      show(terminal);

      syncHandle(terminal, true);
      setActive('[data-toggle-panel="terminal"]', true);

      $$("[data-terminal-tab]").forEach(function (el) {
        var active = el.dataset.terminalTab === name;
        el.classList.toggle("active-terminal", active);
        el.classList.toggle("text-muted-foreground", !active);
      });

      $$("[data-terminal-pane]").forEach(function (el) {
        el.classList.toggle("hidden", el.dataset.terminalPane !== name);
      });

      syncCenterLayout();
      window.dispatchEvent(new Event("resize"));
    }

    function closeTerminal(name) {
      var terminal = panel("terminal");
      var tab = $('[data-terminal-tab="' + name + '"]');
      var pane = $('[data-terminal-pane="' + name + '"]');

      if (!tab) return;

      var wasActive = tab.classList.contains("active-terminal");

      destroyTerminal(name);
      tab.remove();
      if (pane) pane.remove();

      var remaining = visibleTerminalTabs();

      if (!remaining.length) {
        hide(terminal);
        syncHandle(terminal, false);
        setActive('[data-toggle-panel="terminal"]', false);
        syncCenterLayout();
        window.dispatchEvent(new Event("resize"));
        return;
      }

      if (wasActive) {
        openTerminal(remaining[0].dataset.terminalTab);
        return;
      }

      syncCenterLayout();
      window.dispatchEvent(new Event("resize"));
    }

        var tabsForNav = visibleEditorTabs();
        var currentIndex = tabsForNav.findIndex(function (tab) { return tab.classList.contains("active-tab"); });
        if (tabsForNav.length) {
          var delta = action.dataset.editorBack !== undefined ? -1 : 1;
          var nextIndex = (currentIndex + delta + tabsForNav.length) % tabsForNav.length;
          openEditor(tabsForNav[nextIndex].dataset.tab);
        }
        return;
      }
      if (action.dataset.fsNewFile !== undefined) return createFsItem("file", fsTargetDirectory(selectedFsPath, selectedFsType));
      if (action.dataset.fsNewFolder !== undefined) return createFsItem("directory", fsTargetDirectory(selectedFsPath, selectedFsType));
      if (action.dataset.fsUploadButton !== undefined) {
        var uploadInput = $("[data-fs-upload-input]");
        if (uploadInput) uploadInput.click();
        return;
      }
      if (action.dataset.fsRefresh !== undefined) return refreshFsTree();
      if (action.dataset.fsMenuAction) {
        var fsAction = action.dataset.fsMenuAction;
        hideFsContextMenu();
        if (fsAction === "open") {
          var row = $$('.fs-row').find(function (item) { return item.dataset.fsPath === contextFsPath; });
          if (contextFsType === "directory") return row && toggleFsDir(row);
          return openFileInEditor(contextFsPath);
        }
        if (fsAction === "new-file") return createFsItem("file", fsTargetDirectory(contextFsPath, contextFsType));
        if (fsAction === "new-folder") return createFsItem("directory", fsTargetDirectory(contextFsPath, contextFsType));
        if (fsAction === "rename") return renameFsItem(contextFsPath);
        if (fsAction === "delete") return deleteFsItem(contextFsPath);
        if (fsAction === "download") return downloadFsItem(contextFsPath, contextFsType);
      }

      if (action.dataset.reconnectTerminal !== undefined) {
        var activeTerminal = visibleTerminalTabs().find(function (tab) {
          return tab.classList.contains("active-terminal");
        });
        var activeName = activeTerminal && activeTerminal.dataset.terminalTab;
        if (activeName && terminalStates[activeName]) {
          reconnectTerminalState(terminalStates[activeName], root.dataset.instanceId);
        }
        return;
      }

      if (action.dataset.newTerminal !== undefined) {
        var terminalName = createCleanTerminal();
        if (terminalName) openTerminal(terminalName);
        return;
      }

      if (action.classList.contains("tab-close")) {
        var editorTab = action.closest("[data-tab]");
        if (editorTab) closeEditor(editorTab.dataset.tab);
        return;
      }

      if (action.classList.contains("terminal-tab-close")) {
        var terminalTab = action.closest("[data-terminal-tab]");
        if (terminalTab) closeTerminal(terminalTab.dataset.terminalTab);
        return;
      }

      if (action.classList.contains("fs-row")) {
        if (action.dataset.fsInlineCreate !== undefined) return;
        selectFsRow(action);
        if (action.dataset.fsType === "directory") return toggleFsDir(action);
        return openFileInEditor(action.dataset.fsPath);
      }

      if (action.dataset.tab) return openEditor(action.dataset.tab);
      if (action.dataset.terminalTab) return openTerminal(action.dataset.terminalTab);
    });

    function toggleRightPanel() {
      var right = panel("right");
      if (!right) return;
      if (!hidden(right)) {
        hide(right);
        syncHandle(right, false);
        setActive("[data-toggle-right]", false);
      } else {
        var activePane = $$("[data-right-pane]").find(function (el) { return !el.classList.contains("hidden"); });
        openRight(activePane ? activePane.dataset.rightPane : "notes");
      }
      window.dispatchEvent(new Event("resize"));
    }

    function focusTerminal() {
      var terminalName = ensureTerminalTab();
      if (!terminalName) return;
      if (hidden(panel("terminal"))) togglePanel("terminal");
      else openTerminal(terminalName);
      var state = terminalStates[terminalName];
      if (state && state.term) state.term.focus();
    }
